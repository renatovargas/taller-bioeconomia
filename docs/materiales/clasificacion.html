Clasificación de productos y actividades

Qué es bioeconomía, qué es bioeconomía parcial y qué queda fuera, año por año

Cada producto (fila del COU) y cada actividad (columna) tiene una sola categoría, válida para todos los años en que INDEC publica su código. Elija un año para ver qué códigos existen, cuánto pesan en la producción y a qué categoría pertenecen.

function segmented(opciones, valor, etiqueta) {
  const el = html`<div class="seg" role="radiogroup" aria-label=${etiqueta}></div>`;
  el.value = valor;
  const botones = opciones.map(o => {
    const b = html`<button type="button" role="radio">${o}</button>`;
    const marcar = () => botones.forEach(([x, v]) => { x.classList.toggle("on", v === el.value); x.setAttribute("aria-checked", v === el.value); });
    b.addEventListener("click", () => { el.value = o; marcar(); el.dispatchEvent(new Event("input", {bubbles: true})); });
    el.append(b);
    return [b, o];
  });
  botones.forEach(([x, v]) => { x.classList.toggle("on", v === valor); x.setAttribute("aria-checked", v === valor); });
  return el;
}

function chips(cs) {
  const el = html`<div class="chips" aria-label="Categorías"></div>`;
  el.value = cs.map(c => c.cat);
  const bs = cs.map(c => {
    const b = html`<button type="button" aria-pressed="true"><i></i>${c.nombre}</button>`;
    b.className = "chip on c" + c.cat;
    b.addEventListener("click", () => {
      const on = !b.classList.contains("on");
      b.classList.toggle("on", on); b.setAttribute("aria-pressed", on);
      el.value = cs.filter((_, k) => bs[k].classList.contains("on")).map(x => x.cat);
      el.dispatchEvent(new Event("input", {bubbles: true}));
    });
    el.append(b);
    return b;
  });
  return el;
}

function buscador() {
  const el = html`<label class="buscar"><span class="sr-only">Buscar</span><input type="search" placeholder="Buscar: carne, 1514, papel…" autocomplete="off"></label>`;
  const inp = el.querySelector("input");
  el.value = "";
  inp.addEventListener("input", (e) => { e.stopPropagation(); el.value = inp.value; el.dispatchEvent(new Event("input", {bubbles: true})); });
  return el;
}

function mosaico(L, etiquetaGrupo) {
  const vis = L.filter(coincide);
  if (!vis.length) return html`<p class="vacio">Ningún código coincide con la búsqueda y las categorías elegidas.</p>`;
  const max = Math.max(...L.map(d => d.s), 1e-9);
  const grupos = d3.groups(vis.sort((a, b) => a.og - b.og || a.cod.localeCompare(b.cod, "es", {numeric: true})), d => d.grupo);
  return html`<div class="mosaico">${grupos.map(([g, xs]) => {
    const tot = L.filter(d => d.grupo === g);
    return html`<section class="grupo">
      <header><h4>${g}</h4><span class="g-n">${xs.length} de ${tot.length} ${etiquetaGrupo}</span>
        <span class="g-mini">${C.map(c => html`<i class="c${c.cat}" style=${"flex-grow:" + tot.filter(d => d.cat === c.cat).length}></i>`)}</span>
      </header>
      <div class="tiles">${xs.map(d => html`<div class="tile c${d.cat}" tabindex="0">
        <span class="t-cod">${d.cod}${d.nuevo ? html`<b class="nuevo">nuevo</b>` : ""}</span>
        <span class="t-nom">${d.nombre}</span>
        <span class="t-peso" style=${"width:" + Math.max(2, 100 * d.s / max) + "%"}></span>
        <span class="tip"><b>${d.cod}</b> · ${C[d.cat - 1].nombre}<br>${d.completo}<br>
          <em>${fmtPct(d.s)} de la producción de ${anio} · vigente desde ${d.desde}${d.hasta ? " hasta " + d.hasta : ""}${d.serie !== d.cod ? " · serie " + d.serie : ""}</em></span>
      </div>`)}</div>
    </section>`;
  })}</div>`;
}

function tabla(L, etiqueta) {
  const vis = L.filter(coincide).sort((a, b) => b.s - a.s);
  const max = Math.max(...L.map(d => d.s), 1e-9);
  return html`<div class="tabla-scroll"><table class="cou clasif-tabla">
    <thead><tr><th class="izq">Código</th><th class="izq">${etiqueta}</th><th class="izq">Grupo</th><th class="izq">Categoría</th><th>Peso en la producción, ${anio}</th><th>Vigencia</th></tr></thead>
    <tbody>${vis.map(d => html`<tr>
      <td class="izq cod">${d.cod}</td><td class="izq nom">${d.completo}</td><td class="izq">${d.grupo}</td>
      <td class="izq"><span class="pill p${d.cat}">${C[d.cat - 1].nombre}</span></td>
      <td class="peso"><span class="mini-barra"><i class="c${d.cat}" style=${"width:" + (100 * d.s / max) + "%"}></i></span>${fmtPct(d.s)}</td>
      <td>${d.desde}${d.hasta ? "–" + d.hasta : "–"}</td></tr>`)}</tbody></table></div>`;
}

Cada recuadro es una columna del COU (CIIU Rev. 3, 107 actividades en todos los años). Una actividad de una categoría puede producir productos de otra: por eso la cuenta clasifica ambas dimensiones por separado.

Los códigos de INDEC cambian poco, pero cambian. Cada código vigente en un año tiene su color; los años vacíos son años en que el código no aparece en el cuadro. El código serie une códigos que se fusionaron, para comparar entre años.

Además:

  • Unidad. De 2018 a 2022, INDEC publica en miles de pesos y en 2023, en millones. La base usa millones de pesos para todos los años.
  • Alias. En la hoja de utilización, INDEC escribe 15120, 15130, 15140, 15200 para las actividades 1512, 1513, 1514, 1520. El lookup registra ambos códigos.

1 Un renglón por código. arg_lookups.xlsx tiene una hoja para productos y otra para actividades. Cada código de INDEC aparece una sola vez, con su categoría.

2 Vigencia en lugar de versiones. Las columnas Desde y Hasta dicen en qué años el código está en el cuadro. Un código nuevo se agrega con su Desde; uno que desaparece recibe su Hasta.

3 La rutina comprueba. Para cada año, los códigos del cuadro deben coincidir con los vigentes en el lookup, y cada fila debe sumar el total publicado por INDEC. Si algo no cuadra, se detiene y dice qué código falta o sobra.

4 Reclasificar es editar una celda. Las categorías viven en la hoja categorias. Cambiar la condición de un producto en un renglón la cambia en todos los años.

Las tres categorías siguen la terminología del SCN 2025 para cuentas temáticas (capítulo 38). Una cuenta temática se organiza alrededor de una actividad clave con una frontera precisa:

  • Bioeconomía: dentro de la frontera.
  • Bioeconomía parcial: parcialmente dentro. Son filas y columnas con contenido biológico y no biológico que el COU de INDEC no permite separar.
  • Fuera de la bioeconomía: fuera de la frontera.
